        // --- 0. TWINKLING STARFIELD BACKGROUND ENGINE ---
        function initStarfield() {
            const canvas = document.getElementById('starfieldCanvas');
            const ctx = canvas.getContext('2d');
            let stars = [];
            const numStars = 170;

            function resizeCanvas() {
                canvas.width = window.innerWidth;
                canvas.height = window.innerHeight;
                createStars();
            }

            function createStars() {
                stars = [];
                for (let i = 0; i < numStars; i++) {
                    stars.push({
                        x: Math.random() * canvas.width,
                        y: Math.random() * canvas.height,
                        radius: Math.random() * 1.4 + 0.3,
                        baseAlpha: Math.random() * 0.65 + 0.15,
                        twinkleSpeed: Math.random() * 0.02 + 0.005,
                        phase: Math.random() * Math.PI * 2,
                        driftY: (Math.random() - 0.5) * 0.08
                    });
                }
            }

            function animateStars() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);

                for (let s of stars) {
                    s.phase += s.twinkleSpeed;
                    s.y += s.driftY;
                    if (s.y < 0) s.y = canvas.height;
                    if (s.y > canvas.height) s.y = 0;

                    const alpha = Math.max(0.05, Math.min(1, s.baseAlpha + Math.sin(s.phase) * 0.35));
                    ctx.beginPath();
                    ctx.arc(s.x, s.y, s.radius, 0, Math.PI * 2);
                    ctx.fillStyle = `rgba(255, 255, 255, ${alpha.toFixed(3)})`;
                    ctx.fill();

                    // Subtle cross-glint on brighter stars
                    if (s.radius > 1.35 && alpha > 0.7) {
                        ctx.strokeStyle = `rgba(255, 255, 255, ${(alpha * 0.45).toFixed(3)})`;
                        ctx.lineWidth = 0.6;
                        ctx.beginPath();
                        ctx.moveTo(s.x - 4, s.y);
                        ctx.lineTo(s.x + 4, s.y);
                        ctx.moveTo(s.x, s.y - 4);
                        ctx.lineTo(s.x, s.y + 4);
                        ctx.stroke();
                    }
                }
                requestAnimationFrame(animateStars);
            }

            window.addEventListener('resize', resizeCanvas);
            resizeCanvas();
            animateStars();
        }

        function updateTelemetry() {

            const ex = parseInt(inputs.exercise.value);

            labels.switches.textContent = `${sw}/day`;

            // Normalized pillar scores (0 - 100)
            const dwScore = Math.min(100, (dw / 6) * 100);
            const distScore = Math.max(0, 100 - (dist / 6) * 100);
            const slpScore = slp >= 7.5 && slp <= 9 ? 100 : Math.max(0, 100 - Math.abs(8 - slp) * 20);

            const ring = document.getElementById('scoreRing');
            const circumference = 314.16;
            const offset = circumference - (finalScore / 100) * circumference;
            ring.style.strokeDashoffset = offset;

            const badge = document.getElementById('scoreBadge');
            if (finalScore >= 80) {
                ring.style.stroke = '#ffffff';
                badge.textContent = 'Peak Focus';
                badge.className = 'text-xs font-bold px-2.5 py-0.5 rounded-full bg-white text-black border border-white';

                leakEl.textContent = `Sleep Deficit (${slp.toFixed(1)}h logged)`;
                leakEl.className = 'text-sm font-semibold text-neutral-200 mt-0.5';

        // --- 5. GEMINI AI COACHING ENGINE (WITH SMART FALLBACK) ---
        async function runAICoach() {
            const metrics = updateTelemetry();
            const userGoal = inputs.goal.value.trim() || 'Stay focused and productive';

            const loadingEl = document.getElementById('aiLoading');
            const contentEl = document.getElementById('aiContent');
            const statusEl = document.getElementById('aiStatusText');
            const btnEl = document.getElementById('analyzeBtn');

            // Move user straight into the Outcome interface to see live analysis
            switchStage('outcome');

            loadingEl.classList.remove('hidden');
            contentEl.classList.add('opacity-40');
            statusEl.textContent = 'Analyzing...';
            btnEl.disabled = true;

            const prompt = `You are AURA OS, an elite AI life and focus coach. Analyze this user's daily telemetry and provide hyper-specific, practical advice so they stay focused.
User Goal: "${userGoal}"
Metrics:
- Deep Work: ${metrics.dw} hours
- Social Media/Distractions: ${metrics.dist} hours
- Sleep: ${metrics.slp} hours
- Exercise: ${metrics.ex} minutes
- App/Context Switches: ${metrics.sw} per day
- Calculated Focus Score: ${metrics.finalScore}/100`;

            const apiKey = ""; // Canvas runtime injects key automatically
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

            const payload = {
                contents: [{ parts: [{ text: prompt }] }],
                generationConfig: {
                    responseMimeType: "application/json",
                    responseSchema: {
                        type: "OBJECT",
                        properties: {
                            diagnosis: { type: "STRING" },
                            tips: {
                                type: "ARRAY",
                                items: {
                                    type: "OBJECT",
                                    properties: {
                                        tag: { type: "STRING" },
                                        action: { type: "STRING" }
                                    },
                                    required: ["tag", "action"]
                                }
                            },
                            schedule: {
                                type: "ARRAY",
                                items: {
                                    type: "OBJECT",
                                    properties: {
                                        time: { type: "STRING" },
                                        block: { type: "STRING" }
                                    },
                                    required: ["time", "block"]
                                }
                            }
                        },
                        required: ["diagnosis", "tips", "schedule"]
                    }
                }
            };

            try {
                const data = await fetchWithRetry(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(payload)
                });

                const rawText = data?.candidates?.[0]?.content?.parts?.[0]?.text;
                if (!rawText) throw new Error('Empty AI response');

                const parsed = JSON.parse(rawText);
                renderAIAdvice(parsed);
                statusEl.textContent = 'Updated via Gemini AI';
            } catch (err) {
                const fallback = generateLocalFallbackAdvice(metrics, userGoal);
                renderAIAdvice(fallback);
                statusEl.textContent = 'Telemetry Engine';
            } finally {
                loadingEl.classList.add('hidden');
                contentEl.classList.remove('opacity-40');
                btnEl.disabled = false;
            }
        }

        // Exponential backoff helper
        async function fetchWithRetry(url, options, retries = 2, delay = 1000) {
            for (let i = 0; i <= retries; i++) {
                try {
                    const res = await fetch(url, options);
                    if (!res.ok) throw new Error(`HTTP ${res.status}`);
                    return await res.json();
                } catch (e) {
                    if (i === retries) throw e;
                    await new Promise(r => setTimeout(r, delay * Math.pow(2, i)));
                }
            }
        }

        }

        // --- 6. LIVE FOCUS SESSION TIMER ---
        let timerSeconds = 25 * 60;
        let timerInterval = null;

        function toggleTimer() {
            const btn = document.getElementById('timerToggleBtn');
            const pulse = document.getElementById('timerPulse');

            if (timerInterval) {
                clearInterval(timerInterval);
                timerInterval = null;
                btn.textContent = 'Resume Focus';
                btn.className = 'px-4 py-2 rounded-xl bg-white hover:bg-neutral-200 text-black text-xs font-bold transition';
                pulse.className = 'w-3 h-3 rounded-full bg-neutral-400';
            } else {
                btn.textContent = 'Pause';
                btn.className = 'px-4 py-2 rounded-xl bg-neutral-800 hover:bg-neutral-700 text-white border border-neutral-600 text-xs font-bold transition';
                pulse.className = 'w-3 h-3 rounded-full bg-white animate-ping';

                timerInterval = setInterval(() => {
                    if (timerSeconds > 0) {
                        timerSeconds--;
                        updateTimerDisplay();
                    } else {
                        clearInterval(timerInterval);
                        timerInterval = null;
                        inputs.deepWork.value = Math.min(12, parseFloat(inputs.deepWork.value) + 0.5);
                        updateTelemetry();
                        showToast('Focus Sprint Complete! +0.5h added to Deep Work.');
                        resetTimer();
                    }
                }, 1000);
            }
        }

        function resetTimer() {
            clearInterval(timerInterval);
            timerInterval = null;
            timerSeconds = 25 * 60;
            updateTimerDisplay();
            const btn = document.getElementById('timerToggleBtn');
            btn.textContent = 'Start Focus';
            btn.className = 'px-4 py-2 rounded-xl bg-white hover:bg-neutral-200 text-black text-xs font-bold transition';
            document.getElementById('timerPulse').className = 'w-3 h-3 rounded-full bg-neutral-600';
        }

        function updateTimerDisplay() {
            const mins = String(Math.floor(timerSeconds / 60)).padStart(2, '0');
            const secs = String(timerSeconds % 60).padStart(2, '0');
            document.getElementById('timerDisplay').textContent = `${mins}:${secs}`;
        }

                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;');
